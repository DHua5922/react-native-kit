# Input.Number

## Import

```jsx
import { Input } from "react-native-kit";
```

## Examples

```jsx
function Example() {
  const [number, setNumber] = useState("");

  return (
    <Input.Number
      placeholder="Enter number here"
      value={number}
      onChange={(number) => setNumber(number)}
    />
  );
}
```
